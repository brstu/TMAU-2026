82void runInteractive() {
83 std::cout << "Lab 03 / Variant 21 / Closed-loop temperature control\n";
84 auto preset = presets[readValue("Plant [1 = 1.2, 2 = 2.7, 3 = 3.5]: ", 1, 3) - 1];
85 if (readValue("Coefficients [1 = preset, 2 = custom]: ", 1, 2) == 2) {
86 customize(preset);
87 }
88 const auto mode = readValue("Mode [1 = fixed steps, 2 = error tolerance]: ", 1, 2) == 1
89 ? StopMode::fixedSteps : StopMode::errorTolerance;
90 const auto steps = static_cast<std::size_t>(
91 readValue(mode == StopMode::fixedSteps ? "Number of steps: " : "Maximum steps: ", 1, 1000000));
92 const double dt = readValue("Sample period (seconds): ", 1e-9, 1e6);
93 const double initial = readValue("Initial temperature: ", -1e6, 1e6);
94 const double target = readValue("Target temperature: ", -1e6, 1e6);
95 const double tolerance = mode == StopMode::errorTolerance
96 ? readValue("Absolute error tolerance: ", 0.0, 1e6) : 0.0;
97 const auto result = simulate(preset.plant, preset.pid, {mode, steps, dt, initial, target, tolerance});
98 writeCsv("results.csv", result);
99 const auto& lastSample = result.samples.back();
100 std::cout << "step,time,target,output,error,command\n"
101 << lastSample.step << ',' << lastSample.time << ',' << lastSample.target << ',' << lastSample.output
102 << ',' << lastSample.error << ',' << lastSample.command << '\n';
103 switch (result.reason) {
104 case StopReason::completed: std::cout << "Fixed number of steps completed.\n"; break;
105 case StopReason::toleranceReached: std::cout << "Error tolerance reached.\n"; break;
106 case StopReason::stepLimit: std::cout << "Step limit reached before the error tolerance.\n"; break;
107 }
108 std::cout << "Trajectory saved to results.csv\n";
109}

111void writeToleranceExample(const std::filesystem::path& directory, std::ostream& summary,
112 const Preset& preset, int target, double tolerance) {
113 const auto result = simulate(preset.plant, preset.pid,
114 {StopMode::errorTolerance, 200, 0.2, 20.0,
115 static_cast<double>(target), tolerance});
116 const auto& last = result.samples.back();
117 summary << preset.key << ',' << target << ',' << tolerance << ','
118 << last.step << ',' << last.time << ',' << last.output << ','
119 << last.error << ',' << (result.reason == StopReason::toleranceReached
120 ? "toleranceReached" : "stepLimit") << '\n';
121 if (tolerance == 0.2) {
122 writeCsv(directory / (std::string("modeb_") + preset.key + "_" +
123 std::to_string(target) + ".csv"), result);
124 }

127void runExamples() {
128 const std::filesystem::path directory = "build/examples";
129 std::filesystem::create_directories(directory);
130 for (const auto& preset : presets) {
131 for (const int target : {40, 60}) {
132 const auto csv = std::string(preset.key) + "_" + std::to_string(target) + ".csv";
133 const auto result = simulate(preset.plant, preset.pid,
134 {StopMode::fixedSteps, 200, 0.2, 20.0, static_cast<double>(target)});
135 writeCsv(directory / csv, result);
136 }
137 }
138
139 const auto& reference = presets.front();
140 const SimulationSettings tuning{StopMode::fixedSteps, 200, 0.2, 20, 60};
141 auto gains = reference.pid;
142 gains.ki = 0.0;
143 gains.kd = 0.0;
144 writeCsv(directory / "tuning_p.csv", simulate(reference.plant, gains, tuning));
145 gains.ki = reference.pid.ki;
146 writeCsv(directory / "tuning_pi.csv", simulate(reference.plant, gains, tuning));
147 writeCsv(directory / "tuning_pid.csv", simulate(reference.plant, reference.pid, tuning));
148 std::ofstream summary(directory / "modeb_summary.csv");
149 summary.exceptions(std::ios::badbit | std::ios::failbit);
150 summary.imbue(std::locale::classic());
151 summary << "model,target,tolerance,step,time,output,error,reason\n" << std::setprecision(17);
152 for (const auto& preset : presets) {
153 for (const int target : {40, 60}) {
154 for (const double tolerance : {0.5, 0.2, 0.05}) {
155 writeToleranceExample(directory, summary, preset, target, tolerance);
156 }
157 }
158 }
159 summary.close();
160 std::cout << "Example trajectories saved to build/examples\n";
161}
